តារាង teachers តែមួយរួមជាមួយការកំណត់តួនាទី

នៅសាលារៀននេះ មនុស្សធំគ្រប់រូបដែលធ្វើការទីនេះពាក់ផ្លាកឈ្មោះដំបូងដូចគ្នាគឺ៖ "គ្រូបង្រៀន"។ គ្រូបង្រៀនខ្លះក៏កាន់សោបម្រុងផងដែរ — មួយបើកការិយាល័យនាយកសាលា មួយទៀតបើកទូដាក់សៀវភៅបណ្ណាល័យ — ប៉ុន្តែអ្នកត្រូវតែពាក់ផ្លាកឈ្មោះគ្រូបង្រៀនជាមុនសិន ទើបអ្នកត្រូវបានអនុញ្ញាតឱ្យកាន់សោបម្រុងណាមួយបាន។

នៅក្នុងកម្មវិធី "ផ្លាកឈ្មោះ" គឺជารịchដេញដอง (Row) ក្នុងតារាង teachers ដែលគណនីបុគ្គលិកនីមួយៗទទួលបាន ហើយ "សោបម្រុង" គឺជាទង់កំណត់ តួនាទី (role) (admin, librarian) ដែលដាក់ត្រួតលើវា។

"ទូដាក់ឥវ៉ាន់" គឺជារịchដេញដองនៅក្នុងតារាង teachers ហើយ "ឈ្មោះ" គឺជារịchដេញដองនៅក្នុងតារាង users របស់ Supabase ដែលភ្ជាប់គ្នាទៅវិញទៅមកយ៉ាងជាក់លាក់។

មុនពេលសំបុត្រមួយច្បាប់ត្រូវបានបញ្ជូនត្រឡប់ទៅផ្ទះដោយមានត្រាផ្លូវការរបស់សាលា នាយកសាលាត្រូវតែអាននិងចុះហត្ថលេខាលើវាជាមុនសិន។ គ្រូបង្រៀនម្នាក់អាចសរសេរសំបុត្រនោះបាន ប៉ុន្តែគាត់មិនអាចបោះត្រា និងផ្ញើវាចេញដោយខ្លួនឯងបានទេ។

"ត្រា" គឺស្ថានភាព (status) របស់តំណភ្ជាប់របាយការណ៍ដែលប្តូរទៅជា approved (បានអនុម័ត) ហើយ "តុរបស់នាយកសាលា" គឺជាអេក្រង់សម្រាប់ admin ប៉ុណ្ណោះដែលអាចកំណត់វាបាន។

គ្រូបង្រៀនបង្កើតរបាយការណ៍មួយដែលមាន status = 'draft' (សេចក្តីព្រាង); RLS រារាំងកូនសោ anon មិនឱ្យអានរịchដេញដองที่เป็นសេចក្តីព្រាងឡើយ។ មានតែគណនីដែលមានតួនាទីជា admin ប៉ុណ្ណោះដែលអាចប្តូរ status ទៅជា 'approved' — ដែលជាខណៈពេលជាក់លាក់ដែលរịchដេញដองនោះអាចមើលឃើញតាមរយៈគោលការណ៍តំណភ្ជាប់សាធារណៈ។ គ្រូបង្រៀនរៀបចំអ្វីៗគ្រប់យ៉ាង ប៉ុន្តែការផ្លាស់ប្តូរទៅជា "ការផ្សព្វផ្សាយ" ត្រូវបានគ្រប់គ្រងដោយតួនាទីផ្សេងទាំងស្រុង។

ក្រៅបណ្ដាញ (Offline-first) & ការធ្វើសមកាលកម្ម (Syncing)

តើមានអ្វីកើតឡើងនៅពេលអ៊ីនធឺណិតក្នុងបន្ទប់រៀនដាច់?

ក្រដាសស្ទីគ័រនៅលើក្តារខៀនបិតសេចក្តីជូនដំណឹង (Corkboard)

ជួររង់ចាំការសរសេរក្រៅបណ្ដាញ (offline write queue), ច្រោះចេញតាមរបៀប FIFO (ចូលមុន ចេញមុន)

ខ្សែទូរស័ព្ទក្នុងបន្ទប់រៀនដាច់។ ជំនួសឱ្យការបោះបង់ចោល គ្រូបង្រៀនបានសរសេរពិន្ទូនីមួយៗនៅលើក្រដាសស្ទីគ័រដាច់ដោយឡែកពីគ្នា ហើយបិតវានៅលើក្តារខៀនតាមលំដាប់លំដោយដែលគាត់បានសរសេរ។ នៅពេលដែលខ្សែទូរស័ព្ទដំណើរការឡើងវិញ គាត់មិនបានស្រែកប្រាប់វាទាំងអស់គ្នាតែម្តងនោះទេ — គឺគាត់អានវាចេញពីក្តារខៀនពីរសន្លឹក ពីលើចុះក្រោម តាមលំដាប់លំដោយពិតប្រាកដដែលបានបិតវាឡើង។

"ក្តារខៀន" គឺជាជួររង់ចាំក្នុងអង្គចងចាំក្នុងឧបករណ៍ (local queue) សម្រាប់រាល់ការកែប្រែដែលកំពុងរង់ចាំ។ "ការអានវាចេញតាមលំដាប់" គឺជាវិធីសាស្ត្រដែលជួររង់ចាំនោះបញ្ចេញទិន្នន័យ (drains) នៅពេលដែលការតភ្ជាប់អ៊ីនធឺណិតត្រឡប់មកវិញ។

🏫 ការិយាល័យ (បានធ្វើសមកាលកម្ម, តាមលំដាប់លំដោយ)

សរសេរកំណត់ចំណាំបន្តិចបន្តួចពេលក្រៅបណ្ដាញ រួចធ្វើសមកាលកម្ម — មើលពួកវាទៅដល់តាមលំដាប់លំដោយពិតប្រាកដដែលបានសរសេរ។

សរសេរមុន អានមុន

ហេតុអ្វីបានជាជួររង់ចាំមិនអាចផ្លាស់ប្តូរលំដាប់ដោយខ្លួនឯង

ក្រដាសស្ទីគ័រត្រូវបានទាញចេញពីក្តារខៀនក្នុងលំដាប់លំដោយពិតប្រាកដ — សរសេរមុន អានចេញឮៗមុន។ គ្មាននរណាម្នាក់ត្រូវបានអនុញ្ញាតឱ្យទាញយករបស់ពីកណ្តាលគំនរមកស្រែកប្រាប់មុនពេលនោះទេ។

"សរសេរមុន អានមុន" គឺ FIFO (First-In, First-Out — ចូលមុន ចេញមុន) ដែលជាការធានាលំដាប់នៅលើជួររង់ចាំក្រៅបណ្ដាញ។

ជួររង់ចាំត្រូវបានកំណត់ដោយលេខលំដាប់តម្រៀបដោយស្វ័យប្រវត្តិ (auto-incrementing sequence number) ដែលកំណត់នៅពេលសរសេរ មិនមែនពេលធ្វើសមកាលកម្មទេ។ រង្វិលជុំសម្អាត (drain loop) តែងតែស្នើសុំលេខលំដាប់ទាបបំផុតដែលមិនទាន់បានធ្វើសមកាលកម្មជាបន្តបន្ទាប់ និងរង់ចាំឱ្យវាជោគជ័យមុនពេលបន្តទៅមុខទៀត។ នេះមានសារៈសំខាន់សម្រាប់ការសរសេរទិន្នន័យពឹងផ្អែកលើគ្នា — ឧទាហរណ៍ ពិន្ទុដែលយោងទៅលើកំណត់ត្រាវត្តមានដែលបានបង្កើតឡើងប្រាំនាទីមុន — ដែលការចាក់សារថ្មីខុសលំដាប់អាចបំពានលើកូនសោបរទេស ឬសរសេរជាន់លើការកែសម្រួលពេលក្រោយជាមួយនឹងការកែសម្រួលមុន។

ឯកសារថតចម្លងដែលមានត្រា "ប្រើបានត្រឹមតែ ២៤ ម៉ោងប៉ុណ្ណោះ"

ការរក្សាទុកទិន្នន័យអានជាបណ្តោះអាសន្ន (Read cache) ជាមួយ TTL (ថិរវេលាអាយុកាល)

គ្រូបង្រៀនម្នាក់រក្សាទុកឯកសារថតចម្លងបញ្ជីឈ្មោះសិស្សថ្ងៃនេះក្នុងកាបូបរបស់គាត់ ក្នុងករណីដែលការិយាល័យបិទនៅពេលដែលគាត់ត្រូវការវា។ ឯកសារថតចម្លងនោះដំណើរការបានល្អមួយរយៈ — ប៉ុន្តែវាមានត្រាអុចមួយថា៖ "ប្រើបានត្រឹមតែ ២៤ ម៉ោងប៉ុណ្ណោះ"។ បន្ទាប់ពីពេលនោះមក វាចាស់ពេកមិនអាចទុកចិត្តបានដោយមិនបាច់ពិនិត្យម្តងទៀតទេ។

"ឯកសារថតចម្លងដែលមានត្រា" គឺជាទិន្នន័យអានដែលបានរក្សាទុកក្នុងឃ្លាំងទិន្នន័យបណ្តោះអាសន្ន (cache) ជាមួយពេលវេលាទាញយកដែលបានរក្សាទុក។ "ចាស់ពេកមិនអាចទុកចិត្តបាន" គឺជាការផុតកំណត់អាយុកាល (TTL) នៃ cache នោះ។

ប្រសិនបើខ្សែដៃរបស់អ្នកផុតកំណត់ដោយស្ងៀមស្ងាត់ពេលយប់ អ្នកនឹងមិនដឹងរហូតដល់អ្នកដើរទៅមាត់ទ្វារម្តងទៀតហើយវាបន្លឺសំឡេងក្រហម។ អ្នកមើលទៅហាក់ដូចជាពាក់ខ្សែដៃដែលមានសុពលភាពពេញមួយពេល — វាគ្រាន់តែលែងដំណើរការជាក់ស្តែងទៀតប៉ុណ្ណោះ។

"សំឡេងបន្លឺ" គឺជាការឆ្លើយតបកូដ 401 ពី Supabase នៅពេលក្រោយដែលនិមិត្តសញ្ញាត្រូវបានប្រើប្រាស់ពិតប្រាកដទល់នឹងម៉ាស៊ីនមេ (server) ដោយសារតែមិនមានវិធីណាដើម្បីពិនិត្យមើលការផុតកំណត់ពេលក្រៅបណ្ដាញពេញលេញនោះទេ។

JWTs មានការផុតកំណត់ផ្ទាល់ខ្លួន (exp claim) ប៉ុន្តែឧបករណ៍ក្រៅបណ្ដាញមិនអាចសួរ Supabase ដើម្បីផ្ទៀងផ្ទាត់ ឬបន្តវាបានទេ — វាអាចគ្រាន់តែជឿជាក់លើនិមិត្តសញ្ញាដែលវាមានស្រាប់ប៉ុណ្ណោះ។ ជាទូទៅ និមិត្តសញ្ញាបន្ត (refresh token) ต่ออายุសម័យដោយស្ងៀមស្ងាត់មុនពេលផុតកំណត់ ប៉ុន្តែប្រសិនបើឧបករណ៍នៅតែក្រៅបណ្ដាញហួសពេលកំណត់នោះ សំណើដំបូងបន្ទាប់ពីភ្ជាប់ឡើងវិញនឹងបរាជ័យដោយសារកូដ 401 ដែលជំរុញឱ្យ PSMS ព្យាយាមបន្ត និងបើបរាជ័យ នឹងត្រឡប់ទៅអេក្រង់ចូលគណនីពិតប្រាកដវិញ។

ពេលវេលាពិត (Real-time) & ការជូនដំណឹង

តើអ្នកនាយកសាលាដឹងភ្លាមៗនៅពេលមានរឿងអ្វីមួយកើតឡើងយ៉ាងដូចម្ដេច?

កណ្តឹងដែលភ្ជាប់ជាមួយទូសំបុត្រ

ការជាវទិន្នន័យ Supabase Realtime

ជំនួសឱ្យការដើរទៅទូសំបុត្ររាល់ប្រាំនាទីម្តងដើម្បីពិនិត្យរកមើលសំបុត្រថ្មី ទូសំបុត្រមានកណ្តឹងតូចមួយភ្ជាប់ជាមួយវា។ ពេលដែលមានសំបុត្រធ្លាក់ចូលភ្លាម កណ្តឹងក៏រោទ៍ឡើង — គ្មាននរណាម្នាក់ត្រូវដើររកមើលនោះទេ។

"កណ្តឹង" គឺជាការជាវ Supabase Realtime តាមរយៈ WebSocket។ "សំបុត្រធ្លាក់ចូល" គឺជាការបញ្ចូលរịchដេញដองចូលទៅក្នុងតារាងដែលកំពុងត្រូវបានតាមដាន។

PSMS បើកការជាវ supabase.channel(...) ដែលត្រងទៅលើ postgres_changes លើតារាង notifications សម្រាប់ school_id បច្ចុប្បន្ន។ រាល់ពេលដែលរịchដេញដងដែលត្រូវគ្នាត្រូវបានបញ្ចូល ស្ទ្រីមការចម្លងទិន្នន័យរបស់ Supabase ជំរុញទិន្នន័យទៅកាន់គ라이언트ផ្ទាល់តាមរយៈ WebSocket ដែលបើកស្រាប់ — គ្មានរង្វិលជុំស្វែងរកទិន្នន័យដដែលៗ (polling loop) គ្មានការស្នើសុំ SELECT ដដែលៗរាល់ពីរបីវិនាទីដែលស៊ីថ្មនិងកម្រិតបញ្ជូនអ៊ីនធឺណិតឡើយ។

នរណាជាអ្នករោទ៍កណ្តឹង?

ตัวกระตุ้นฐานข้อมูล (Database trigger)

កណ្តឹងត្រូវបានភ្ជាប់ដោយផ្ទាល់ទៅនឹងទូសំបុត្រផ្ទាល់ មិនមែនភ្ជាប់ទៅនឹងអ្នកបញ្ជូនសំបុត្រណាម្នាក់ឡើយ។ ដូច្នេះវាទោះបីជាអ្នកណាជាអ្នកបញ្ជូនសំបុត្រក៏ដោយ ក៏វារោទ៍ដែរ — មិនថាអ្នកបញ្ជូនធម្មតា អ្នកជំនួស ឬនរណាម្នាក់យកមកដាក់ដោយផ្ទាល់ដៃទេ។

"ការភ្ជាប់ខ្សែ" គឺជាຕົວกระตุ้น (trigger) របស់ Postgres លើតារាងដើម។ "មិនថាអ្នកណាជាអ្នកបញ្ជូន" មានន័យថាការជូនដំណឹងដំណើរការដោយមិនគិតថាកម្មវិធីផ្នែកណាមួយជាអ្នកសរសេរទិន្នន័យនោះទេ។

ฟังก์ชัน AFTER INSERT trigger บนตารางอย่าง grades ឬ attendance ដំណើរការមុខងារដែលបញ្ចូលរịchដេញដងដែលត្រូវគ្នាទៅក្នុង notifications។ ដោយសារតែកម្មវិធីกระตุ้นនេះស្ថិតនៅក្នុងមូលដ្ឋានទិន្នន័យជំនួសឱ្យកូដរបស់គ라이언트ណាមួយ ពិន្ទុដែលបានបញ្ចូលពីកម្មវិធីมือถือ កម្មវិធីเว็บ ឬสคริปต์นำเข้าของผู้ดูแลระบบในอนาคត ล้วនបង្កើតការជូនដំណឹងដូចគ្នា — ការធានានេះស្ថិតនៅក្រោមស្រទាប់កម្មវិធី (application layer)។

PWA & Service Worker

ហេតុអ្វីបានជាកម្មវិធីមានអារម្មណ៍ថាលឿនរហ័ស ទោះបីជាក្រៅបណ្ដាញក៏ដោយ?

អ្នកជំនួយការដ៏ចិត្តល្អដែលចាំថា តើសៀវភៅស្ថិតនៅឯណា

Service worker (កម្មវិធីជំនួយផ្ទៃខាងក្រោយ)

សូមស្រមៃមើលអ្នកជំនួយការដ៏ចិត្តល្អម្នាក់ដែលឈរនៅចន្លោះអ្នកនិងបណ្ណាល័យសាលារៀន។ លើកដំបូងដែលអ្នកសុំសៀវភៅ ពួកគេដើរទៅយកវា ប៉ុន្តែពួកគេចាំយ៉ាងច្បាស់ថាវាស្ថិតនៅឯណា — ដូច្នេះលើកក្រោយ ពួកគេប្រគល់វាឱ្យអ្នកភ្លាមៗដោយមិនបាច់ដើរទៅទីនោះម្តងទៀតទេ។

"អ្នកជំនួយការ" គឺ service worker — ដែលជាស្គ្រីបស្ទាក់ចាប់សំណើបណ្តាញ (network requests) "ការចាំថាវាស្ថិតនៅឯណា" គឺជាឃ្លាំងទិន្នន័យបណ្តោះអាសន្នសម្រាប់ឯកសារដែលធ្លាប់ទាញយកមកហើយ។

PSMS ចុះឈ្មោះ service worker ដែលស្ទាក់ចាប់ព្រឹត្តិការណ៍ fetch សម្រាប់ទ្រព្យសម្បត្តិថេររបស់ខ្លួន (ชุดคำสั่ง JS, ไอคอน, ฟอนต์) និងឃ្លាំងទិន្នន័យបណ្តោះអាសន្នពួកវាក្នុងការផ្ទុកលើកដំបូងតាមរយៈ Cache API ការផ្ទុកលើកក្រោយនឹងបម្រើការដោយตรงពីឃ្លាំងទិន្នន័យបណ្តោះអាសន្នទាំងនោះមុនពេលពិនិត្យមើលបណ្តាញទៅទៀត — ស្រោមកម្មវិធីបង្ហាញឡើងភ្លាមៗ ទោះបីជាក្នុងការតភ្ជាប់ที่ไม่เสถียរ ខណៈដែលទិន្នន័យពិតប្រាកដនៅតែឆ្លងកាត់ប្រព័ន្ធឃ្លាំងអាន និងជួររង់ចាំការសរសេរដាច់ដោយឡែក។

អ្នកជំនួយការរង់ចាំរហូតដល់អ្នកនិយាយថា "យល់ព្រម"

ជូនដំណឹងអំពីការធ្វើបច្ចុប្បន្នភាព មិនមែនធ្វើបច្ចុប្បន្នភាពដោយស្វ័យប្រវត្តិទេ

អ្នកប្រាកដជាមិនចង់ឱ្យអ្នកជំនួយការបណ្ណាល័យប្តូរសៀវភៅក្នុងដៃរបស់អ្នកទៅជាច្បាប់ថ្មីជាងនេះ ខណៈពេលដែលអ្នកកំពុងអានពាក់កណ្តាលប្រយោគនោះទេ។ ពួកគេរង់ចាំរហូតដល់អ្នកនិយាយថា "យល់ព្រម ខ្ញុំរួចរាល់ហើយ" មុនពេលបញ្ជូនច្បាប់ចម្លងថ្មីឱ្យ។

"ច្បាប់ចម្លងថ្មី" គឺជាកំណែកម្មវិធីដែលเพิ่งปรับใช้ใหม่ "ការរង់ចាំពាក្យថា យល់ព្រម" គឺជាការជូនដំណឹងធ្វើបច្ចុប្បន្នភាពដែលបង្ហាញឡើងជំនួសឱ្យការบังคับให้ផ្ទុកឡើងវិញភ្លាមៗ។

កំណែ service worker ថ្មីដំឡើងនៅផ្ទៃខាងក្រោយប៉ុន្តែនៅតែស្ថិតក្នុងស្ថានភាព waiting ជំនួសឱ្យការបើកដំណើរការភ្លាមៗ — ការបើកដំណើរការនឹងបំផ្លាញសំណើដែលកំពុងដំណើរការ និងស្ថានភាពក្នុងเครื่องកណ្តាលគំនិត។ PSMS ស្តាប់ព្រឹត្តិការណ៍នេះ និងបង្ហាញការជូនដំណឹង "មានបច្ចុប្បន្នភាពថ្មី" តូចមួយ; មានតែពេលអ្នកប្រើប្រាស់ប៉ះវាប៉ុណ្ណោះ ទើបកម្មវិធីហៅ skipWaiting() និងផ្ទុកឡើងវិញ ដែលធានាថាការធ្វើបច្ចុប្បន្នភាពមិនរំខាននរណាម្នាក់ក្នុងពេលកំពុងกรอกข้อมูล។

លុបមុខងារអាជីវកម្ម (Business Logic)

តើកម្មវិធីសម្រេចចិត្តបង្ហាញអ្វីខ្លះយ៉ាងដូចម្ដេច?

ក្មេងពីរនាក់ចូលដល់បន្ទាត់ទីបញ្ចប់ជាមួយគ្នាទទួលបានចំណាត់ថ្នាក់លេខមួយរួមគ្នា

ការគណនាចំណាត់ថ្នាក់ដោយការរំលងលេខ (Rank computation with tie-skipping)

ក្មេងពីរនាក់រត់ចូលបន្ទាត់ទីបញ្ចប់ក្នុងពេលជាមួយគ្នាเป๊ะ ពួកគេទទួលបានខ្សែបូ "លេខ 1" ទាំងពីរនាក់។ ដោយសារតែមានក្មេងពីរនាក់គ្រងចំណាត់ថ្នាក់លេខ 1 និងលេខ 2 ទៅហើយ ក្មេងបន្ទាប់ក្នុងជួរនឹងទទួលបានខ្សែបូ "លេខ 3" — នឹងគ្មានថ្ងៃទទួលបាន "លេខ 2" ដាច់ខាត។

"រំលងទៅលេខ 3 ភ្លាម" គឺជាสิ่งที่មុខងាររៀបចំចំណាត់ថ្នាក់ដ៏ត្រឹមត្រូវធ្វើនៅពេលមានពិន្ទុស្មើគ្នា ជំនួសឱ្យការគ្រាន់តែរាប់ទីតាំងម្តងមួយៗ។

ការរៀបចំចំណាត់ថ្នាក់ថ្នាក់រៀនប្រើប្រាស់คำสั่ง RANK() OVER (ORDER BY average_score DESC) របស់ Postgres ជំនួសឱ្យ ROW_NUMBER()។ RANK() ផ្តល់តម្លៃដូចគ្នាដល់រịchដេញដងដែលស្មើគ្នា និងរំលងចំណាត់ថ្នាក់បន្ទាប់តាមភាពសមស្រប (1, 1, 3) ដែលត្រូវគ្នាទៅនឹងពិធីប្រគល់ខ្សែបូពិតប្រាកដ — ខណៈពេលដែល ROW_NUMBER() នឹងកាត់លទ្ធផលដោយចៃដន្យជានิจ និងចែក "លេខ 2" ដែលមិនสะท้อนការពិត។
